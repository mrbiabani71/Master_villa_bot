    Returns a multiline text with villa view statistics.
    """
    villas = get_all_villas()
    if not villas:

    lines = []

        if vid is None:

        count = get_stat(key) or 0

        name = villa.get("title") or villa.get("name") or f"ویلا {vid}"
        lines.append(f"- {name}: {count} بازدید")

    return "\n".join(lines)
